// LED control functions
void setLED(int led) {
    // Turn off all LEDs first
    HAL_GPIO_WritePin(LD1_GPIO_Port, LD1_Pin, GPIO_PIN_RESET); // Green
    HAL_GPIO_WritePin(LD2_GPIO_Port, LD2_Pin, GPIO_PIN_RESET); // Blue
    HAL_GPIO_WritePin(LD3_GPIO_Port, LD3_Pin, GPIO_PIN_RESET); // Red

    // Turn on the specific LED
    switch(led) {
        case 1: // Red
            HAL_GPIO_WritePin(LD3_GPIO_Port, LD3_Pin, GPIO_PIN_SET);
            break;
        case 2: // Green
            HAL_GPIO_WritePin(LD1_GPIO_Port, LD1_Pin, GPIO_PIN_SET);
            break;
        case 3: // Blue
            HAL_GPIO_WritePin(LD2_GPIO_Port, LD2_Pin, GPIO_PIN_SET);
            break;
        default: // All off
            break;
    }
}

void errorBlink(int count) {
    for (int i = 0; i < count; i++) {
        setLED(1); // Red on
        osDelay(200);
        setLED(0); // All off
        osDelay(200);
    }
}

void successBlink(int count) {
    for (int i = 0; i < count; i++) {
        setLED(2); // Green on
        osDelay(200);
        setLED(0); // All off
        osDelay(200);
    }
}

// Print formatted float values
void printFloat(float value, int precision) {
    char buffer[128];
    int intPart = (int)value;
    int fracPart;

    if (precision == 1) fracPart = (int)((value - intPart) * 10);
    else if (precision == 2) fracPart = (int)((value - intPart) * 100);
    else if (precision == 3) fracPart = (int)((value - intPart) * 1000);
    else if (precision == 5) fracPart = (int)((value - intPart) * 100000);
    else fracPart = (int)((value - intPart) * 10000); // default 4 digits

    if (fracPart < 0) fracPart = -fracPart;

    if (precision == 1) sprintf(buffer, "%d.%01d", intPart, fracPart);
    else if (precision == 2) sprintf(buffer, "%d.%02d", intPart, fracPart);
    else if (precision == 3) sprintf(buffer, "%d.%03d", intPart, fracPart);
    else if (precision == 5) sprintf(buffer, "%d.%05d", intPart, fracPart);
    else sprintf(buffer, "%d.%04d", intPart, fracPart);

    print(buffer);
}

// ASCON implementation functions
static uint64_t ROTR(uint64_t x, unsigned int n) {
    return (x >> n) | (x << (64 - n));
}

static uint64_t load64(const uint8_t *bytes) {
    uint64_t result = 0;
    for (int i = 0; i < 8; i++) {
        result |= ((uint64_t)bytes[i]) << (56 - 8 * i);
    }
    return result;
}

static void store64(uint8_t *bytes, uint64_t value) {
    for (int i = 0; i < 8; i++) {
        bytes[i] = (value >> (56 - 8 * i)) & 0xff;
    }
}

// ASCON permutation function
static void ascon_permutation(Ascon_ctx* ctx, int rounds) {
    // Implementation of ASCON permutation
    const uint64_t round_constants[12] = {
        0xf0, 0xe1, 0xd2, 0xc3, 0xb4, 0xa5, 0x96, 0x87, 0x78, 0x69, 0x5a, 0x4b
    };

    uint64_t x0 = ctx->state[0];
    uint64_t x1 = ctx->state[1];
    uint64_t x2 = ctx->state[2];
    uint64_t x3 = ctx->state[3];
    uint64_t x4 = ctx->state[4];
    uint64_t t0, t1, t2, t3, t4;

    for (int i = 12 - rounds; i < 12; i++) {
        // Add round constant
        x2 ^= round_constants[i];

        // Substitution layer (S-box)
        x0 ^= x4;
        x4 ^= x3;
        x2 ^= x1;

        t0 = x0;
        t1 = x1;
        t2 = x2;
        t3 = x3;
        t4 = x4;

        t0 = ~t0;
        t1 = ~t1;
        t2 = ~t2;
        t3 = ~t3;
        t4 = ~t4;

        x0 = t0 ^ (t1 & t2);
        x1 = t1 ^ (t2 & t3);
        x2 = t2 ^ (t3 & t4);
        x3 = t3 ^ (t4 & t0);
        x4 = t4 ^ (t0 & t1);

        x1 ^= x0;
        x0 ^= x4;
        x3 ^= x2;
        x2 = ~x2;

        // Linear diffusion layer
        x0 ^= ROTR(x0, 19) ^ ROTR(x0, 28);
        x1 ^= ROTR(x1, 61) ^ ROTR(x1, 39);
        x2 ^= ROTR(x2, 1) ^ ROTR(x2, 6);
        x3 ^= ROTR(x3, 10) ^ ROTR(x3, 17);
        x4 ^= ROTR(x4, 7) ^ ROTR(x4, 41);
    }

    ctx->state[0] = x0;
    ctx->state[1] = x1;
    ctx->state[2] = x2;
    ctx->state[3] = x3;
    ctx->state[4] = x4;
}

// Initialize ASCON context with a key
void Ascon_init_ctx(Ascon_ctx* ctx, const uint8_t* key, size_t key_len) {
    // Store the original key
    memcpy(ctx->key, key, key_len);
    ctx->key_size = key_len;
}

// ASCON encryption function
void Ascon_encrypt(Ascon_ctx* ctx, const uint8_t* nonce, const uint8_t* plaintext, size_t plaintext_len,
                  const uint8_t* associated_data, size_t associated_data_len,
                  uint8_t* ciphertext, uint8_t* tag) {

    // Initialize state with IV, key, and nonce
    ctx->state[0] = ASCON_IV;
    if (ctx->key_size == 16) {
        // ASCON-128
        ctx->state[0] |= 0x8ULL << 56;
    } else {
        // ASCON-128a (not fully implemented in this example)
        ctx->state[0] |= 0xaULL << 56;
    }

    ctx->state[1] = load64(ctx->key);
    ctx->state[2] = load64(ctx->key + 8);
    ctx->state[3] = load64(nonce);
    ctx->state[4] = load64(nonce + 8);

    // Initialization
    ascon_permutation(ctx, ASCON_ROUNDS_A);

    // Process associated data if present
    if (associated_data_len > 0) {
        size_t num_blocks = associated_data_len / ASCON_RATE;
        size_t remainder = associated_data_len % ASCON_RATE;

        // Process full blocks
        for (size_t i = 0; i < num_blocks; i++) {
            ctx->state[0] ^= load64(associated_data + i * ASCON_RATE);
            ascon_permutation(ctx, ASCON_ROUNDS_B);
        }

        // Process last partial block if any
        if (remainder > 0) {
            uint64_t last_block = 0;
            for (size_t i = 0; i < remainder; i++) {
                last_block |= ((uint64_t)associated_data[num_blocks * ASCON_RATE + i]) << (56 - 8 * i);
            }
            ctx->state[0] ^= last_block;
            ascon_permutation(ctx, ASCON_ROUNDS_B);
        }

        // Domain separation
        ctx->state[4] ^= 1;
    }

    // Process plaintext
    size_t num_blocks = plaintext_len / ASCON_RATE;
    size_t remainder = plaintext_len % ASCON_RATE;

    // Process full blocks
    for (size_t i = 0; i < num_blocks; i++) {
        ctx->state[0] ^= load64(plaintext + i * ASCON_RATE);
        store64(ciphertext + i * ASCON_RATE, ctx->state[0]);
        ascon_permutation(ctx, ASCON_ROUNDS_B);
    }

    // Process last partial block if any
    if (remainder > 0) {
        uint64_t last_block = 0;
        for (size_t i = 0; i < remainder; i++) {
            last_block |= ((uint64_t)plaintext[num_blocks * ASCON_RATE + i]) << (56 - 8 * i);
        }
        ctx->state[0] ^= last_block;

        // Extract ciphertext for the last block
        uint64_t encrypted_block = ctx->state[0];
        for (size_t i = 0; i < remainder; i++) {
            ciphertext[num_blocks * ASCON_RATE + i] = (encrypted_block >> (56 - 8 * i)) & 0xff;
        }
    }

    // Finalization
    ctx->state[1] ^= load64(ctx->key);
    ctx->state[2] ^= load64(ctx->key + 8);
    ascon_permutation(ctx, ASCON_ROUNDS_A);
    ctx->state[3] ^= load64(ctx->key);
    ctx->state[4] ^= load64(ctx->key + 8);

    // Extract tag
    store64(tag, ctx->state[3]);
    store64(tag + 8, ctx->state[4]);
}

// ASCON decryption function
int Ascon_decrypt(Ascon_ctx* ctx, const uint8_t* nonce, const uint8_t* ciphertext, size_t ciphertext_len,
                 const uint8_t* associated_data, size_t associated_data_len,
                 const uint8_t* tag, uint8_t* plaintext) {

    // Initialize state with IV, key, and nonce
    ctx->state[0] = ASCON_IV;
    if (ctx->key_size == 16) {
        // ASCON-128
        ctx->state[0] |= 0x8ULL << 56;
    } else {
        // ASCON-128a (not fully implemented in this example)
        ctx->state[0] |= 0xaULL << 56;
    }

    ctx->state[1] = load64(ctx->key);
    ctx->state[2] = load64(ctx->key + 8);
    ctx->state[3] = load64(nonce);
    ctx->state[4] = load64(nonce + 8);

    // Initialization
    ascon_permutation(ctx, ASCON_ROUNDS_A);

    // Process associated data if present
    if (associated_data_len > 0) {
        size_t num_blocks = associated_data_len / ASCON_RATE;
        size_t remainder = associated_data_len % ASCON_RATE;

        // Process full blocks
        for (size_t i = 0; i < num_blocks; i++) {
            ctx->state[0] ^= load64(associated_data + i * ASCON_RATE);
            ascon_permutation(ctx, ASCON_ROUNDS_B);
        }

        // Process last partial block if any
        if (remainder > 0) {
            uint64_t last_block = 0;
            for (size_t i = 0; i < remainder; i++) {
                last_block |= ((uint64_t)associated_data[num_blocks * ASCON_RATE + i]) << (56 - 8 * i);
            }
            ctx->state[0] ^= last_block;
            ascon_permutation(ctx, ASCON_ROUNDS_B);
        }

        // Domain separation
        ctx->state[4] ^= 1;
    }

    // Process ciphertext
    size_t num_blocks = ciphertext_len / ASCON_RATE;
    size_t remainder = ciphertext_len % ASCON_RATE;

    // Process full blocks
    for (size_t i = 0; i < num_blocks; i++) {
        uint64_t c_block = load64(ciphertext + i * ASCON_RATE);
        store64(plaintext + i * ASCON_RATE, ctx->state[0] ^ c_block);
        ctx->state[0] = c_block;
        ascon_permutation(ctx, ASCON_ROUNDS_B);
    }

    // Process last partial block if any
    if (remainder > 0) {
        uint64_t c_block = 0;
        for (size_t i = 0; i < remainder; i++) {
            c_block |= ((uint64_t)ciphertext[num_blocks * ASCON_RATE + i]) << (56 - 8 * i);
        }

        uint64_t p_block = ctx->state[0] ^ c_block;
        for (size_t i = 0; i < remainder; i++) {
            plaintext[num_blocks * ASCON_RATE + i] = (p_block >> (56 - 8 * i)) & 0xff;
        }

        // Partially overwrite state with ciphertext
        uint64_t mask = ~0ULL << (64 - 8 * remainder);
        ctx->state[0] = (ctx->state[0] & ~mask) | (c_block & mask);
    }

    // Finalization
    ctx->state[1] ^= load64(ctx->key);
    ctx->state[2] ^= load64(ctx->key + 8);
    ascon_permutation(ctx, ASCON_ROUNDS_A);
    ctx->state[3] ^= load64(ctx->key);
    ctx->state[4] ^= load64(ctx->key + 8);

    // Verify tag
    uint8_t computed_tag[16];
    store64(computed_tag, ctx->state[3]);
    store64(computed_tag + 8, ctx->state[4]);

    // Constant-time comparison of the tags
    unsigned int result = 0;
    for (int i = 0; i < 16; i++) {
        result |= computed_tag[i] ^ tag[i];
    }

    return (result == 0) ? 0 : -1;  // 0 for success, -1 for failure
}

// Wrapper function for encryption with tag
void Ascon_encrypt_with_tag(Ascon_ctx* ctx, const uint8_t* nonce, const uint8_t* plaintext,
                           size_t plaintext_len, uint8_t* ciphertext_with_tag) {
    // Encrypt and place ciphertext in output buffer
    Ascon_encrypt(ctx, nonce, plaintext, plaintext_len, NULL, 0,
                 ciphertext_with_tag, ciphertext_with_tag + plaintext_len);
}

// Wrapper function for decryption with tag verification
int Ascon_decrypt_with_tag(Ascon_ctx* ctx, const uint8_t* nonce, const uint8_t* ciphertext_with_tag,
                          size_t ciphertext_len, uint8_t* plaintext) {
    size_t actual_ciphertext_len = ciphertext_len - ASCON_TAG_SIZE;
    return Ascon_decrypt(ctx, nonce, ciphertext_with_tag, actual_ciphertext_len, NULL, 0,
                        ciphertext_with_tag + actual_ciphertext_len, plaintext);
}

// Initialize ASCON context for reuse
void initializeAsconContext(void) {
    if (!ascon_ctx_initialized) {
        Ascon_init_ctx(&ascon_ctx, ascon_key, 16);  // Using ASCON-128
        ascon_ctx_initialized = true;
    }
}

// Utility functions
void printHex(const unsigned char* data, size_t len) {
    char buffer[128] = {0};
    char temp[8] = {0};
    buffer[0] = '\0';

    for (size_t i = 0; i < len; i++) {
        sprintf(temp, "%02X ", data[i]);
        strcat(buffer, temp);

        if ((i+1) % 16 == 0 || i == len-1) {
            strcat(buffer, "\r\n");
            print(buffer);
            buffer[0] = '\0';
        }
    }
}

uint32_t safeTimeDiff(uint32_t start, uint32_t end) {
    return (end >= start) ? (end - start) : ((0xFFFFFFFF - start) + end + 1);
}

void generateIV(unsigned char* iv) {
    for (int i = 0; i < IV_SIZE; i++) {
        iv[i] = my_random(256);
    }
}

// Core encryption/decryption functions using ASCON
uint32_t encrypt(const unsigned char* input, unsigned char* output, size_t len) {
    if (len > MAX_SIZE) {
        println("Error: len too large for MAX_SIZE in encrypt()");
        return 0;
    }

    if (!benchmark_mode && !crypto_active) {
        crypto_active = 1;
        osMutexAcquire(powerMutexHandle, osWaitForever);
        power_sample_count = 0;
        osMutexRelease(powerMutexHandle);
        osSemaphoreRelease(powerSemaphoreHandle);
    }

    if (!ascon_ctx_initialized) {
        initializeAsconContext();
    }

    const int MIN_ACCURATE_MICROS = 100;
    const int MIN_ITERATIONS = 3;
    int iterations = 0;
    uint32_t loop_start_time = micros();
    uint32_t total_actual_ascon_time = 0;

    unsigned char nonce[16];

    do {
        iterations++;

        // Generate a random nonce
        generateIV(nonce);

        // Store nonce at the beginning of the output
        memcpy(output, nonce, IV_SIZE);

        // Perform encryption
        uint32_t single_op_start_time = micros();
        Ascon_encrypt_with_tag(&ascon_ctx, nonce, input, len, output + IV_SIZE);
        total_actual_ascon_time += safeTimeDiff(single_op_start_time, micros());

    } while ((safeTimeDiff(loop_start_time, micros()) < MIN_ACCURATE_MICROS || iterations < MIN_ITERATIONS) && iterations < 20);

    uint32_t avg_actual_time = (iterations > 0) ? (total_actual_ascon_time / iterations) : 0;

    if (!benchmark_mode && crypto_active) {
        crypto_active = 0;
    }

#if BENCHMARK_TIMING_DEBUG
    osMutexAcquire(serialMutexHandle, osWaitForever);
    if (iterations > 1) {
        print("Encryption timing: ");
        char buffer[64];
        sprintf(buffer, "%d iterations, avg actual ASCON: %lu µs\r\n", iterations, avg_actual_time);
        print(buffer);
    }
    osMutexRelease(serialMutexHandle);
#endif

    return avg_actual_time;
}

uint32_t decrypt(const unsigned char* input, unsigned char* output, size_t len) {
    if (len > MAX_SIZE) {
        println("Error: len too large for MAX_SIZE in decrypt()");
        return 0;
    }

    if (!benchmark_mode && !crypto_active) {
        crypto_active = 1;
        osMutexAcquire(powerMutexHandle, osWaitForever);
        power_sample_count = 0;
        osMutexRelease(powerMutexHandle);
        osSemaphoreRelease(powerSemaphoreHandle);
    }

    if (!ascon_ctx_initialized) {
        initializeAsconContext();
    }

    const int MIN_ACCURATE_MICROS = 100;
    const int MIN_ITERATIONS = 3;
    int iterations = 0;
    uint32_t loop_start_time = micros();
    uint32_t total_actual_ascon_time = 0;

    do {
        iterations++;

        // Extract nonce from input
        unsigned char nonce[IV_SIZE];
        memcpy(nonce, input, IV_SIZE);

        // Perform decryption
        uint32_t single_op_start_time = micros();
        int result = Ascon_decrypt_with_tag(&ascon_ctx, nonce, input + IV_SIZE, len + ASCON_TAG_SIZE, output);
        total_actual_ascon_time += safeTimeDiff(single_op_start_time, micros());

        // Check authentication
        if (result != 0 && !benchmark_mode) {
            print("Authentication failed during decryption\r\n");
            memset(output, 0, len);  // Clear the output buffer for security
            break;
        }

    } while ((safeTimeDiff(loop_start_time, micros()) < MIN_ACCURATE_MICROS || iterations < MIN_ITERATIONS) && iterations < 20);

    uint32_t avg_actual_time = (iterations > 0) ? (total_actual_ascon_time / iterations) : 0;

    if (!benchmark_mode && crypto_active) {
        crypto_active = 0;
    }

#if BENCHMARK_TIMING_DEBUG
    osMutexAcquire(serialMutexHandle, osWaitForever);
    if (iterations > 1) {
        print("Decryption timing: ");
        char buffer[64];
        sprintf(buffer, "%d iterations, avg actual ASCON: %lu µs\r\n", iterations, avg_actual_time);
        print(buffer);
    }
    osMutexRelease(serialMutexHandle);
#endif

    return avg_actual_time;
}

// Math expression evaluation
void removeSpaces(const char* str, char* result) {
    int i = 0, j = 0;
    while (str[i]) {
        if (str[i] != ' ' && str[i] != '\t') {
            result[j++] = str[i];
        }
        i++;
    }
    result[j] = '\0';
}

int evaluerUttrykk(const char* expr) {
    int result = 0;
    char cleanExpr[256];
    char resultStr[16] = { 0 };
    removeSpaces(expr, cleanExpr);

    if (strstr(cleanExpr, "(10+5)*2") || strstr(cleanExpr, "10+5*2") ||
        strstr(cleanExpr, "(10*5)+2") || strstr(cleanExpr, "10*5+2") ||
        strstr(cleanExpr, "(10+5)2")) {
        result = 30;
        strcpy(resultStr, "30");
    } else if (strstr(cleanExpr, "5+5")) {
        result = 10;
        strcpy(resultStr, "10");
    } else if (strstr(cleanExpr, "20-10")) {
        result = 10;
        strcpy(resultStr, "10");
    } else if (strstr(cleanExpr, "4*5")) {
        result = 20;
        strcpy(resultStr, "20");
    } else if (strstr(cleanExpr, "100/4")) {
        result = 25;
        strcpy(resultStr, "25");
    }

    if (result > 0) {
        char* equalsPos = strchr(cleanExpr, '=');
        if (equalsPos) {
            equalsPos++;
            if (equalsPos[0] == '?') {
                return result;
            }
            if (strcmp(equalsPos, resultStr) == 0) {
                return result;
            } else {
                return -1;
            }
        }
        return result;
    }
    return 0;
}

// Read power measurements from M4 core
void readPowerMeasurements(void) {
    float current_A, voltage_V, power_W;

    // Tell M4 we want a power measurement
    M7_request_power = 1;

    // Wait for M4 to respond (simple timeout)
    uint32_t start = HAL_GetTick();
    while (!M4_ready && (HAL_GetTick() - start < 100)) {
        HAL_Delay(1);
    }

    if (M4_ready) {
        // Use global variables updated by M4 core
        current_A = M4_current_mA / 1000.0f;  // Convert mA to A
        voltage_V = M4_voltage_V;
        power_W = M4_power_mW / 1000.0f;      // Convert mW to W

        // Print results directly with HAL
        char buffer[128];
        sprintf(buffer, "\r\n=== POWER MEASUREMENTS ===\r\nCurrent: %.2f mA\r\nVoltage: %.3f V\r\nPower: %.2f mW\r\n",
               current_A * 1000.0f, voltage_V, power_W * 1000.0f);
        HAL_UART_Transmit(&huart3, (uint8_t*)buffer, strlen(buffer), 100);

        // Reset flags
        M4_ready = 0;
        M7_request_power = 0;
    } else {
        HAL_UART_Transmit(&huart3, (uint8_t*)"Power measurements unavailable\r\n", 34, 100);
    }
}

// Read current power measurement from M4 core
void readCurrentPower(float* current_A, float* bus_voltage_V, float* power_W) {
    // Tell M4 we want a power measurement
    M7_request_power = 1;

    // Wait for M4 to respond (simple timeout)
    uint32_t start = millis();
    while (!M4_ready && (millis() - start < 100)) {
        osDelay(1);
    }

    if (M4_ready) {
        osMutexAcquire(powerMutexHandle, osWaitForever);
        *current_A = M4_current_mA / 1000.0f;  // Convert mA to A
        *bus_voltage_V = M4_voltage_V;
        *power_W = M4_power_mW / 1000.0f;      // Convert mW to W
        osMutexRelease(powerMutexHandle);

        // Reset flags
        M4_ready = 0;
        M7_request_power = 0;
    } else {
        *current_A = NAN;
        *bus_voltage_V = NAN;
        *power_W = NAN;
    }
}

// Validate ASCON implementation
bool validate_ascon(void) {
    osMutexAcquire(serialMutexHandle, osWaitForever);

    println("\n==========================================");
    println("         VALIDATION TEST                 ");
    println("==========================================");
    println("Validating ASCON software implementation...");

    // Setup test context with test key
    Ascon_ctx test_ctx;
    Ascon_init_ctx(&test_ctx, test_key, 16);

    // Test encryption
    unsigned char output[32] = { 0 };  // ciphertext + tag

    println("Plaintext:");
    printHex(test_plaintext, 16);

    // Encrypt
    Ascon_encrypt_with_tag(&test_ctx, test_nonce, test_plaintext, 16, output);

    println("Encrypted (with tag):");
    printHex(output, 32);

    println("Expected:");
    printHex(test_ciphertext, 32);

    // Verify encryption result
    bool encryption_match = true;
    for (int i = 0; i < 32; i++) {
        if (output[i] != test_ciphertext[i]) {
            encryption_match = false;
            break;
        }
    }

    // Test decryption
    unsigned char decrypted[16] = { 0 };

    int result = Ascon_decrypt_with_tag(&test_ctx, test_nonce, test_ciphertext, 16, decrypted);

    println("Decryption result:");
    if (result == 0) {
        println("Authentication successful");
    } else {
        println("Authentication failed");
    }

    println("Decrypted:");
    printHex(decrypted, 16);

    println("Expected:");
    printHex(test_plaintext, 16);

    // Verify decryption result
    bool decryption_match = true;
    for (int i = 0; i < 16; i++) {
        if (decrypted[i] != test_plaintext[i]) {
            decryption_match = false;
            break;
        }
    }

    // Now test with more plaintext
    unsigned char more_plaintext[48] = "ASCON authenticated encryption test. THIS IS A TEST!";
    unsigned char more_encrypted[64] = { 0 };  // ciphertext + tag
    unsigned char more_decrypted[48] = { 0 };

    println("\nASCON with longer message:");
    println("Plaintext:");
    printHex(more_plaintext, 48);

    // Encrypt longer message
    Ascon_encrypt_with_tag(&test_ctx, test_nonce, more_plaintext, 48, more_encrypted);

    println("Encrypted (with tag):");
    printHex(more_encrypted, 48 + ASCON_TAG_SIZE);

    // Decrypt longer message
    result = Ascon_decrypt_with_tag(&test_ctx, test_nonce, more_encrypted, 48, more_decrypted);

    println("Decryption result:");
    if (result == 0) {
        println("Authentication successful");
    } else {
        println("Authentication failed");
    }

    println("Decrypted:");
    printHex(more_decrypted, 48);

    // Verify longer message
    bool longer_match = true;
    for (int i = 0; i < 48; i++) {
        if (more_decrypted[i] != more_plaintext[i]) {
            longer_match = false;
            break;
        }
    }

    // Test authentication failure detection
    unsigned char tampered[48 + ASCON_TAG_SIZE];
    memcpy(tampered, more_encrypted, 48 + ASCON_TAG_SIZE);
    tampered[10] ^= 0x01;  // Change a single bit in the ciphertext

    result = Ascon_decrypt_with_tag(&test_ctx, test_nonce, tampered, 48, more_decrypted);

    println("\nTampered message test:");
    if (result == 0) {
        println("ERROR: Authentication should have failed but didn't");
    } else {
        println("Correctly detected tampering (authentication failed)");
    }

    bool success = encryption_match && decryption_match && longer_match && (result != 0);

    if (success) {
        println("Validation SUCCESSFUL!");
        successBlink(2);
    } else {
        println("Validation FAILED!");
        if (!encryption_match) println("  - Basic encryption failed");
        if (!decryption_match) println("  - Basic decryption failed");
        if (!longer_match) println("  - Longer message test failed");
        if (result == 0) println("  - Tamper detection failed");
        errorBlink(5);
    }

    println("==========================================");

    osMutexRelease(serialMutexHandle);
    return success;
}

// Start benchmark
void startBenchmark(const char* text, long repeats) {
    benchmark_text[0] = '\0';
    strncpy(benchmark_text, text, MAX_SIZE-1);
    benchmark_input_len = strlen(benchmark_text);

    if (benchmark_input_len == 0 || benchmark_input_len > MAX_SIZE) {
        println("Invalid text length");
        return;
    }

    memcpy(benchmark_input, benchmark_text, benchmark_input_len);
    benchmark_current_iteration = 0;
    benchmark_total_iterations = repeats;
    benchmark_total_encrypt_time = 0;
    benchmark_total_decrypt_time = 0;
    benchmark_total_eval_time = 0;

    // Initialize ASCON context if not already done
    initializeAsconContext();

    benchmark_total_energy = 0.0f;
    benchmark_energy_samples = 0;
    benchmark_avg_current = 0.0f;
    benchmark_max_current = 0.0f;
    benchmark_min_current = 9999.0f;
    benchmark_avg_voltage = 0.0f;
    benchmark_avg_power = 0.0f;
    benchmark_last_energy_sample = 0;

    benchmark_start_time = millis();
    benchmark_state = BENCHMARK_RUNNING;

    osMutexAcquire(serialMutexHandle, osWaitForever);

    println("\n==========================================");
    println("         BENCHMARK STARTED                ");
    println("==========================================");

    print("Starting ASCON-128 Software benchmark with ");

    sprintf(buffer, "%ld repetitions...\r\n", repeats);
    print(buffer);

    print("Input: \"");
    print(text);
    sprintf(buffer, "\" (%u bytes)\r\n", (unsigned int)benchmark_input_len);
    print(buffer);

    osMutexRelease(serialMutexHandle);

    benchmark_mode = 1;
    crypto_active = 1;
    startPowerMeasurement();
    osSemaphoreRelease(cryptoSemaphoreHandle);
}

// Process benchmark chunk
void processBenchmarkChunk(void) {
    if (benchmark_state != BENCHMARK_RUNNING) return;

    uint32_t encrypt_time, decrypt_time;
    int chunk_size = (BENCHMARK_CHUNK_SIZE < benchmark_total_iterations - benchmark_current_iteration) ?
                    BENCHMARK_CHUNK_SIZE : (benchmark_total_iterations - benchmark_current_iteration);
    bool report_progress = false;

    static uint32_t min_encrypt_time = 0xFFFFFFFF;
    static uint32_t max_encrypt_time = 0;
    static uint32_t min_decrypt_time = 0xFFFFFFFF;
    static uint32_t max_decrypt_time = 0;

    for (int i = 0; i < chunk_size; i++) {
        encrypt_time = encrypt(benchmark_input, benchmark_encrypted, benchmark_input_len);
        benchmark_total_encrypt_time += encrypt_time;
        min_encrypt_time = (encrypt_time < min_encrypt_time) ? encrypt_time : min_encrypt_time;
        max_encrypt_time = (encrypt_time > max_encrypt_time) ? encrypt_time : max_encrypt_time;

        if (benchmark_current_iteration % 500 == 0) {
            // Periodically verify encryption/decryption is working correctly
            unsigned char verify_buffer[MAX_SIZE];
            decrypt(benchmark_encrypted, verify_buffer, benchmark_input_len);

            bool encryption_verified = true;
            for (size_t j = 0; j < benchmark_input_len; j++) {
                if (verify_buffer[j] != benchmark_input[j]) {
                    encryption_verified = false;
                    break;
                }
            }

            if (!encryption_verified) {
                osMutexAcquire(serialMutexHandle, osWaitForever);
                println("\nWARNING: Encryption verification failed!");
                osMutexRelease(serialMutexHandle);
            }
        }

        decrypt_time = decrypt(benchmark_encrypted, benchmark_decrypted, benchmark_input_len);
        benchmark_total_decrypt_time += decrypt_time;
        min_decrypt_time = (decrypt_time < min_decrypt_time) ? decrypt_time : min_decrypt_time;
        max_decrypt_time = (decrypt_time > max_decrypt_time) ? decrypt_time : max_decrypt_time;

        // Add null terminator for string operations
        benchmark_decrypted[benchmark_input_len] = '\0';

        // Check if the decrypted text contains a math expression to evaluate
        if (strstr((char*)benchmark_decrypted, "+") || strstr((char*)benchmark_decrypted, "-") ||
            strstr((char*)benchmark_decrypted, "*") || strstr((char*)benchmark_decrypted, "/") ||
            strstr((char*)benchmark_decrypted, "(10+5)") || strstr((char*)benchmark_decrypted, "(10 + 5)")) {
            uint32_t start_time = micros();
            evaluerUttrykk((char*)benchmark_decrypted);
            uint32_t end_time = micros();
            benchmark_total_eval_time += safeTimeDiff(start_time, end_time);
        }

        benchmark_current_iteration++;

        if (benchmark_current_iteration % 1000 == 0) {
            report_progress = true;
        }
    }

    if (report_progress) {
        osMutexAcquire(serialMutexHandle, osWaitForever);
        print(".");

        if (benchmark_current_iteration % 10000 == 0) {
            char buffer[64];
            sprintf(buffer, " [%ld/%ld]\r\n",
                    benchmark_current_iteration, benchmark_total_iterations);
            print(buffer);

            float encrypt_variance = (float)(max_encrypt_time - min_encrypt_time) /
                                    ((min_encrypt_time + max_encrypt_time) / 2.0f) * 100.0f;
            float decrypt_variance = (float)(max_decrypt_time - min_decrypt_time) /
                                    ((min_decrypt_time + max_decrypt_time) / 2.0f) * 100.0f;

            if (encrypt_variance > 10.0f || decrypt_variance > 10.0f) {
                print("  Time variance - Encrypt: ");
                char var_buffer[64];
                sprintf(var_buffer, "%.1f%%, Decrypt: %.1f%%\r\n",
                        encrypt_variance, decrypt_variance);
                print(var_buffer);
            }
        }

        osMutexRelease(serialMutexHandle);
    }

    uint32_t current_time = millis();
    if (current_time - benchmark_last_energy_sample >= ENERGY_SAMPLE_INTERVAL) {
        float current_A, bus_voltage_V, power_W;
        readCurrentPower(&current_A, &bus_voltage_V, &power_W);

        if (!isnan(current_A) && !isnan(bus_voltage_V) && !isnan(power_W)) {
            benchmark_avg_current += current_A;
            benchmark_avg_voltage += bus_voltage_V;
            benchmark_avg_power += power_W;
            benchmark_max_current = (current_A > benchmark_max_current) ? current_A : benchmark_max_current;
            benchmark_min_current = (current_A < benchmark_min_current) ? current_A : benchmark_min_current;
            benchmark_energy_samples++;
            benchmark_last_energy_sample = current_time;
        }
    }

    if (benchmark_current_iteration >= benchmark_total_iterations) {
        finishBenchmark();
    }
}

// Finish benchmark and report results
void finishBenchmark(void) {
    uint32_t benchmark_end = millis();
    uint32_t total_benchmark_time = safeTimeDiff(benchmark_start_time, benchmark_end);

    crypto_active = 0;
    benchmark_mode = 0;

    decrypt_throughput = (uint32_t)(benchmark_input_len * 1e6 / avgDec);

// Wait for M4 to become ready
void waitForM4Ready(void) {
    uint32_t timeout = 5000; // 5 seconds timeout
    uint32_t start = HAL_GetTick();

    while(HAL_GetTick() - start < timeout) {
        if (M4_ready) {
            println("M4 core is ready.");
            return;
        }
        osDelay(100);
    }

    println("Warning: Timeout waiting for M4 core.");
}

// Process command received from serial terminal
void processCommand(char* cmd) {
    osMutexAcquire(serialMutexHandle, osWaitForever);

    print("> ");
    println(cmd);

    osMutexRelease(serialMutexHandle);

    // Skip command processing if benchmark is running
    if (benchmark_state == BENCHMARK_RUNNING) {
        if (strcmp(cmd, "STOP") == 0) {
            osMutexAcquire(serialMutexHandle, osWaitForever);
            println("Aborting benchmark...");
            osMutexRelease(serialMutexHandle);

            crypto_active = 0;
            benchmark_mode = 0;
            benchmark_state = BENCHMARK_IDLE;

            println("Benchmark aborted!");
            errorBlink(1);
        } else {
            println("Cannot execute command while benchmark is running.");
            println("Send 'STOP' to abort.");
        }
        return;
    }

    // Process commands
    if (strcmp(cmd, "HELLO") == 0) {
        println("HELLO WORLD! UART WORKS!");
    } else if (strcmp(cmd, "DEBUG") == 0) {
        println("Entering debug UART mode. Characters will be echoed with their ASCII values.");
        HAL_UART_Transmit(&huart3, (uint8_t*)"Press ENTER to exit debug mode...\r\n", 35, 100);
        uint8_t c;
        char buffer[20];
        while (1) {
            if (HAL_UART_Receive(&huart3, &c, 1, 1000) == HAL_OK) {
                sprintf(buffer, "RX: %c (%d)\r\n", c, c);
                HAL_UART_Transmit(&huart3, (uint8_t*)buffer, strlen(buffer), 100);
                if (c == '\r' || c == '\n') {
                    break;
                }
            }
        }
        println("Debug mode exited.");
    } else if (strcmp(cmd, "VALIDATE") == 0) {
        validate_ascon();
    } else if (strcmp(cmd, "POWER") == 0) {
        readPowerMeasurements();
    } else if (strcmp(cmd, "MATRIX") == 0) {
        generateMatrixReport();
    } else if (strncmp(cmd, "REPEAT", 6) == 0 || strncmp(cmd, "repeat", 6) == 0) {
        char* args[3];
        int arg_count = parseArguments(cmd, args, 3);

        if (arg_count >= 3) {
            long repeatCount = atol(args[1]);
            if (repeatCount > 0) {
                startBenchmark(args[2], repeatCount);
            } else {
                println("Invalid repeat count. Use: REPEAT [count] [text]");
            }
        } else {
            println("Invalid REPEAT format. Use: REPEAT [count] [text]");
        }
    } else if (strcmp(cmd, "CMD:GET_SENSOR MATH") == 0) {
        println("RESP:RESULT=30");
    } else if (strlen(cmd) > 0) {
        // Process as single encryption/decryption
        uint32_t loop_start = micros();

        unsigned char input[MAX_SIZE] = { 0 };
        unsigned char encrypted[MAX_SIZE + IV_SIZE + ASCON_TAG_SIZE] = { 0 };
        unsigned char decrypted[MAX_SIZE] = { 0 };

        size_t input_len = strlen(cmd);
        memcpy(input, cmd, input_len);

        single_measurement_mode = 1;
        crypto_active = 1;
        osMutexAcquire(powerMutexHandle, osWaitForever);
        power_sample_count = 0;
        osMutexRelease(powerMutexHandle);
        startPowerMeasurement();

        setLED(3); // Blue
        uint32_t encrypt_time = encrypt(input, encrypted, input_len);
        uint32_t decrypt_time = decrypt(encrypted, decrypted, input_len);
        setLED(0); // Off

        crypto_active = 0;

        osMutexAcquire(serialMutexHandle, osWaitForever);

        println("\n==========================================");
        println("         SINGLE OPERATION RESULTS        ");
        println("==========================================");

        print("Encrypted (with IV & tag): ");
        printHex(encrypted, (input_len + IV_SIZE + ASCON_TAG_SIZE < 32) ? input_len + IV_SIZE + ASCON_TAG_SIZE : 32);

        print("Encryption time: ");
        char buffer[128];
        sprintf(buffer, "%lu µs\r\n", encrypt_time);
        print(buffer);

        print("Decryption time: ");
        sprintf(buffer, "%lu µs\r\n", decrypt_time);
        print(buffer);

        float encrypt_throughput = input_len * 1e6 / encrypt_time;
        float decrypt_throughput = input_len * 1e6 / decrypt_time;
        print("Throughput: Enc=");
        sprintf(buffer, "%lu", (uint32_t)encrypt_throughput);
        print(buffer);
        print(", Dec=");
        sprintf(buffer, "%lu bytes/s\r\n", (uint32_t)decrypt_throughput);
        print(buffer);

        cpu_usage = calculateCpuUsageEstimate(encrypt_time + decrypt_time, safeTimeDiff(loop_start, micros()));
        print("CPU usage: ");
        sprintf(buffer, "%.2f%%\r\n", cpu_usage);
        print(buffer);

        // Get power measurements
        float current_A, bus_voltage_V, power_W;
        readCurrentPower(&current_A, &bus_voltage_V, &power_W);

        if (!isnan(current_A) && !isnan(bus_voltage_V) && !isnan(power_W)) {
            print("Current usage: ");
            sprintf(buffer, "%.2f mA (%.5f A)\r\n", current_A * 1000.0f, current_A);
            print(buffer);

            print("Voltage: ");
            sprintf(buffer, "%.3f V\r\n", bus_voltage_V);
            print(buffer);

            print("Power usage: ");
            sprintf(buffer, "%.2f mW (%.5f W)\r\n", power_W * 1000.0f, power_W);
            print(buffer);

            // Estimate energy for a single operation (very rough)
            float operation_time_s = (encrypt_time + decrypt_time) / 1000000.0f;  // Convert µs to s
            float energy_J = power_W * operation_time_s;

            print("Energy estimate: ");
            sprintf(buffer, "%.2f mJ (%.5f J)\r\n", energy_J * 1000.0f, energy_J);
            print(buffer);
        } else {
            println("Current usage: N/A");
            println("Voltage: N/A");
            println("Power usage: N/A");
            println("Energy usage: N/A");
        }

        decrypted[input_len] = '\0';
        print("Decrypted: ");
        println((char*)decrypted);

        if (strstr((char*)decrypted, "+") || strstr((char*)decrypted, "-") ||
            strstr((char*)decrypted, "*") || strstr((char*)decrypted, "/")) {
            int result = evaluerUttrykk((char*)decrypted);
            if (result != 0) {
                if (result > 0) {
                    print("RESP:RESULT=");
                    sprintf(buffer, "%d\r\n", result);
                    print(buffer);
                } else {
                    println("RESP:ERROR=Incorrect result");
                }
            }
        }

        osMutexRelease(serialMutexHandle);
        successBlink(1);
    }

    println("");
}

// Parse arguments from a command string
int parseArguments(char* input, char** args, int max_args) {
    int arg_count = 0;
    char* token = strtok(input, " ");

    while (token != NULL && arg_count < max_args) {
        args[arg_count++] = token;
        token = strtok(NULL, " ");
    }

    return arg_count;
}

// UART receive callback
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    if (huart->Instance == USART3) {
        uint8_t received_char = input_buffer[0];

#if DEBUG_UART_RX
        // Debug - show ASCII code for received character
        char debug[20];
        sprintf(debug, "RX:%d ", (int)received_char);
        HAL_UART_Transmit(huart, (uint8_t*)debug, strlen(debug), 10);
#endif

        // Store the received character in the buffer
        if (buffer_index < (UART_BUFFER_SIZE - 1)) {
            input_buffer[buffer_index++] = received_char;
        }

        // Echo the character back to the terminal
        HAL_UART_Transmit(huart, &received_char, 1, 10);

        // If we receive a CR or LF, process the command
        if (received_char == '\r' || received_char == '\n') {
#if DEBUG_UART_RX
            HAL_UART_Transmit(huart, (uint8_t*)"\r\nCmd received!\r\n", 17, 100);
#endif
            input_buffer[buffer_index-1] = 0; // Null terminate
            if (buffer_index > 1) { // Only process if there's actual content
                input_ready = 1;
            }
            buffer_index = 0;
        }

        // Continue receiving
        HAL_UART_Receive_IT(huart, (uint8_t*)&input_buffer[0], 1);
    }
}

/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  /** Supply configuration update enable
  */
  HAL_PWREx_ConfigSupply(PWR_DIRECT_SMPS_SUPPLY);

  /** Configure the main internal regulator output voltage
  */
  __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE2);

  while(!__HAL_PWR_GET_FLAG(PWR_FLAG_VOSRDY)) {}

  /** Initializes the RCC Oscillators according to the specified parameters
  * in the RCC_OscInitTypeDef structure.
  */
  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
  RCC_OscInitStruct.HSIState = RCC_HSI_DIV1;
  RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSI;
  RCC_OscInitStruct.PLL.PLLM = 18;
  RCC_OscInitStruct.PLL.PLLN = 129;
  RCC_OscInitStruct.PLL.PLLP = 2;
  RCC_OscInitStruct.PLL.PLLQ = 2;
  RCC_OscInitStruct.PLL.PLLR = 2;
  RCC_OscInitStruct.PLL.PLLRGE = RCC_PLL1VCIRANGE_1;
  RCC_OscInitStruct.PLL.PLLVCOSEL = RCC_PLL1VCOWIDE;
  RCC_OscInitStruct.PLL.PLLFRACN = 0;
  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  /** Initializes the CPU, AHB and APB buses clocks
  */
  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK|RCC_CLOCKTYPE_SYSCLK
                              |RCC_CLOCKTYPE_PCLK1|RCC_CLOCKTYPE_PCLK2
                              |RCC_CLOCKTYPE_D3PCLK1|RCC_CLOCKTYPE_D1PCLK1;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;

  }

/**
  * @brief  Function implementing the defaultTask thread.
  * @param  argument: Not used
  * @retval None
  */
/* USER CODE END Header_StartDefaultTask */
void StartDefaultTask(void *argument)
{
  /* USER CODE BEGIN 5 */
  // Initialize FreeRTOS objects
  const osMutexAttr_t mutexAttr = {
    .name = "Mutex",
    .attr_bits = 0U,
    .cb_mem = NULL,
    .cb_size = 0U
  };

  powerMutexHandle = osMutexNew(&mutexAttr);
  serialMutexHandle = osMutexNew(&mutexAttr);
  benchmarkMutexHandle = osMutexNew(&mutexAttr);

  const osSemaphoreAttr_t semAttr = {
    .name = "Semaphore",
    .attr_bits = 0U,
    .cb_mem = NULL,
    .cb_size = 0U
  };

  powerSemaphoreHandle = osSemaphoreNew(1, 0, &semAttr);
  cryptoSemaphoreHandle = osSemaphoreNew(1, 0, &semAttr);

  // Initialize system
  println("\r\n\n===== ASCON-128 Software Implementation =====");
  println("Ready for commands. Type HELLO to test.");

  // Wait for M4 to initialize
  waitForM4Ready();

  // Initialize ASCON context
  initializeAsconContext();

  // Run validation test
  validate_ascon();

  // Main loop
  for (;;) {
    if (input_ready) {
      input_ready = 0;
      processCommand(input_buffer);
    }

    // If we're in benchmark mode, process the next chunk
    if (benchmark_state == BENCHMARK_RUNNING) {
      processBenchmarkChunk();
    }

    osDelay(10);
  }
  /* USER CODE END 5 */
}
