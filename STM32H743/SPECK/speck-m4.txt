// Helper functions for I2C communication with INA226
bool INA226_Init(void) {
  // Read manufacturer ID to check if INA226 is responsive
  uint16_t man_id = INA226_ReadReg(INA226_REG_MANUFACTURER_ID);
  if (man_id != 0x5449) { // TI manufacturer ID
    return false;
  }

  // Calculate calibration value
  uint16_t cal_value = INA226_CalculateCalibrationValue(SHUNT_RESISTOR_VALUE, MAX_CURRENT);

  // Configure INA226
  uint16_t config = INA226_AVG_16 |       // 16 samples averaging
                   INA226_VBUS_1100uS |   // 1.1ms bus voltage conversion time
                   INA226_VSH_1100uS |    // 1.1ms shunt voltage conversion time
                   INA226_MODE_SHUNT_BUS_CONT; // Continuous shunt and bus measurements

  INA226_WriteReg(INA226_REG_CONFIG, config);
  INA226_SetCalibration(cal_value);

  return true;
}

uint16_t INA226_ReadReg(uint8_t reg) {
  uint8_t data[2];
  HAL_I2C_Mem_Read(&hi2c1, (INA226_I2C_ADDRESS << 1), reg, I2C_MEMADD_SIZE_8BIT, data, 2, 100);
  return (data[0] << 8) | data[1];
}

void INA226_WriteReg(uint8_t reg, uint16_t value) {
  uint8_t data[2];
  data[0] = (value >> 8) & 0xFF;
  data[1] = value & 0xFF;
  HAL_I2C_Mem_Write(&hi2c1, (INA226_I2C_ADDRESS << 1), reg, I2C_MEMADD_SIZE_8BIT, data, 2, 100);
}

float INA226_GetShuntVoltage_mV(void) {
  int16_t value = (int16_t)INA226_ReadReg(INA226_REG_SHUNTVOLTAGE);
  return value * 0.0025f; // LSB is 2.5 µV
}

float INA226_GetBusVoltage_V(void) {
  uint16_t value = INA226_ReadReg(INA226_REG_BUSVOLTAGE);
  return value * 0.00125f; // LSB is 1.25 mV
}

float INA226_GetCurrent_mA(void) {
  int16_t value = (int16_t)INA226_ReadReg(INA226_REG_CURRENT);
  return value * 0.1f; // Depends on calibration, assuming LSB = 0.1 mA
}
float INA226_GetPower_mW(void) {
  uint16_t value = INA226_ReadReg(INA226_REG_POWER);
  return value * 2.5f; // Depends on calibration, assuming LSB = 2.5 mW
}

void INA226_SetCalibration(uint16_t cal_value) {
  ina226_calibration_value = cal_value;
  INA226_WriteReg(INA226_REG_CALIBRATION, cal_value);
}

uint16_t INA226_CalculateCalibrationValue(float r_shunt, float i_max) {
  // r_shunt in ohms, i_max in amps
  uint16_t cal_value = (uint16_t)((0.00512f) / (r_shunt * i_max / 32768.0f));
  return cal_value;
}

void ReadPowerMeasurements(void) {
  if (ina226_available) {
    float current_mA = INA226_GetCurrent_mA();
    float voltage_V = INA226_GetBusVoltage_V();
    float power_mW = INA226_GetPower_mW();

    // Update shared memory variables
    M4_current_mA = current_mA;
    M4_voltage_V = voltage_V;
    M4_power_mW = power_mW;
    M4_ready = 1;
  }
}
