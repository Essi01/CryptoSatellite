// Timing functions
uint32_t micros() {
    // Convert milliseconds to microseconds and add the SysTick value
    return (HAL_GetTick() * 1000 + (SysTick->LOAD - SysTick->VAL) / (SystemCoreClock / 1000000));
}

    return HAL_GetTick();
}

    return (uint32_t)((float)rand() / RAND_MAX * max);
}

// LED control functions
void setLED(int led) {
    // Turn off all LEDs first
    HAL_GPIO_WritePin(LD1_GPIO_Port, LD1_Pin, GPIO_PIN_RESET); // Green
    HAL_GPIO_WritePin(LD2_GPIO_Port, LD2_Pin, GPIO_PIN_RESET); // Blue
    HAL_GPIO_WritePin(LD3_GPIO_Port, LD3_Pin, GPIO_PIN_RESET); // Red

    // Turn on the specific LED
    switch(led) {
        case 1: // Red
            HAL_GPIO_WritePin(LD3_GPIO_Port, LD3_Pin, GPIO_PIN_SET);
            break;
        case 2: // Green
            HAL_GPIO_WritePin(LD1_GPIO_Port, LD1_Pin, GPIO_PIN_SET);
            break;
        case 3: // Blue
            HAL_GPIO_WritePin(LD2_GPIO_Port, LD2_Pin, GPIO_PIN_SET);
            break;
        default: // All off
            break;
    }
}

void errorBlink(int count) {
    for (int i = 0; i < count; i++) {

        osDelay(200);
    }
}

// Print formatted float values
void printFloat(float value, int precision) {
    char buffer[128];
    int intPart = (int)value;
    int fracPart;

    if (precision == 1) fracPart = (int)((value - intPart) * 10);
    else if (precision == 2) fracPart = (int)((value - intPart) * 100);
    else if (precision == 3) fracPart = (int)((value - intPart) * 1000);
    else if (precision == 5) fracPart = (int)((value - intPart) * 100000);
    else fracPart = (int)((value - intPart) * 10000); // default 4 digits

    if (fracPart < 0) fracPart = -fracPart;

    if (precision == 1) sprintf(buffer, "%d.%01d", intPart, fracPart);
    else if (precision == 2) sprintf(buffer, "%d.%02d", intPart, fracPart);
    else if (precision == 3) sprintf(buffer, "%d.%03d", intPart, fracPart);
    else if (precision == 5) sprintf(buffer, "%d.%05d", intPart, fracPart);
    else sprintf(buffer, "%d.%04d", intPart, fracPart);

    print(buffer);
}

// AES ECB mode encryption
HAL_StatusTypeDef AES_ECB_Encrypt(const uint8_t* plaintext, uint8_t* ciphertext, uint32_t size) {
    CRYP_ConfigTypeDef config;
    uint32_t temp_size;
    
    // Make sure size is a multiple of 16 bytes (AES block size)
    temp_size = (size % 16 == 0) ? size : ((size / 16) + 1) * 16;
    
    // Configure for AES ECB encryption
    config.DataType = CRYP_DATATYPE_8B;
    config.KeySize = CRYP_KEYSIZE_128B;
    config.pKey = (uint32_t*)test_key;  // Using test key
    config.Algorithm = CRYP_AES_ECB;
    config.pInitVect = NULL;  // Not used in ECB mode
    config.Header = NULL;
    config.HeaderSize = 0;
    config.B0 = NULL;
    config.DataWidthUnit = CRYP_DATAWIDTHUNIT_BYTE;
    config.KeyIVConfigSkip = CRYP_KEYIVCONFIG_ALWAYS;
    
    HAL_CRYP_SetConfig(&hcryp, &config);
    
    // Allocate a buffer for the padded plaintext if needed
    uint8_t* padded_plaintext = NULL;
    if (size % 16 != 0) {
        padded_plaintext = (uint8_t*)malloc(temp_size);
        if (!padded_plaintext) return HAL_ERROR;
        
        // Copy plaintext to padded buffer
        memcpy(padded_plaintext, plaintext, size);
        
        // Add PKCS#7 padding
        uint8_t padding_value = 16 - (size % 16);
        for (uint32_t i = size; i < temp_size; i++) {
            padded_plaintext[i] = padding_value;
        }
        
        return HAL_CRYP_Encrypt(&hcryp, (uint32_t*)padded_plaintext, temp_size, (uint32_t*)ciphertext, HAL_MAX_DELAY);
    }
    
    return HAL_CRYP_Encrypt(&hcryp, (uint32_t*)plaintext, size, (uint32_t*)ciphertext, HAL_MAX_DELAY);
}

uint32_t safeTimeDiff(uint32_t start, uint32_t end) {
    return (end >= start) ? (end - start) : ((0xFFFFFFFF - start) + end + 1);
}

// Core encryption/decryption functions using AES hardware
uint32_t encrypt(const unsigned char* input, unsigned char* output, size_t len) {
    if (len > MAX_SIZE) {
        println("Error: len too large for MAX_SIZE in encrypt()");
        return 0;
    }

    if (!benchmark_mode && !crypto_active) {
        crypto_active = 1;
        osMutexAcquire(powerMutexHandle, osWaitForever);
        power_sample_count = 0;
        osMutexRelease(powerMutexHandle);
        osSemaphoreRelease(powerSemaphoreHandle);
    }

    const int MIN_ACCURATE_MICROS = 100;
    const int MIN_ITERATIONS = 3;
    int iterations = 0;
    uint32_t loop_start_time = micros();
    uint32_t total_actual_aes_time = 0;

    unsigned char iv[IV_SIZE];

    do {
        iterations++;

        // Generate a random IV
        generateIV(iv);

        // Store IV at the beginning of the output
        memcpy(output, iv, IV_SIZE);

        // Perform encryption
        uint32_t single_op_start_time = micros();
        
        // Calculate actual size with padding
        uint32_t padded_len = ((len + 15) / 16) * 16; // Round up to nearest multiple of 16
        HAL_StatusTypeDef status = AES_CBC_Encrypt(input, output + IV_SIZE, len, iv);
        
        if (status != HAL_OK) {
            println("AES CBC Encryption failed!");
            break;
        }
        
        total_actual_aes_time += safeTimeDiff(single_op_start_time, micros());

    } while ((safeTimeDiff(loop_start_time, micros()) < MIN_ACCURATE_MICROS || iterations < MIN_ITERATIONS) && iterations < 20);

    uint32_t avg_actual_time = (iterations > 0) ? (total_actual_aes_time / iterations) : 0;

    if (!benchmark_mode && crypto_active) {
        crypto_active = 0;
    }

#if BENCHMARK_TIMING_DEBUG
    osMutexAcquire(serialMutexHandle, osWaitForever);
    if (iterations > 1) {
        print("Encryption timing: ");
        char buffer[64];
        sprintf(buffer, "%d iterations, avg actual AES: %lu µs\r\n", iterations, avg_actual_time);
        print(buffer);
    }
    osMutexRelease(serialMutexHandle);
#endif

    return avg_actual_time;
}

uint32_t decrypt(const unsigned char* input, unsigned char* output, size_t len) {
    if (len > MAX_SIZE) {
        println("Error: len too large for MAX_SIZE in decrypt()");
        return 0;
    }

    if (!benchmark_mode && !crypto_active) {
        crypto_active = 1;
        osMutexAcquire(powerMutexHandle, osWaitForever);
        power_sample_count = 0;
        osMutexRelease(powerMutexHandle);
        osSemaphoreRelease(powerSemaphoreHandle);
    }

    const int MIN_ACCURATE_MICROS = 100;
    const int MIN_ITERATIONS = 3;
    int iterations = 0;
    uint32_t loop_start_time = micros();
    uint32_t total_actual_aes_time = 0;

    do {
        iterations++;

        // Extract IV from input
        unsigned char iv[IV_SIZE];
        memcpy(iv, input, IV_SIZE);

        // Calculate actual ciphertext length with padding
        size_t padded_len = ((len + 15) / 16) * 16; // Round up to nearest multiple of 16
        
        // Perform decryption
        uint32_t single_op_start_time = micros();
        
        HAL_StatusTypeDef status = AES_CBC_Decrypt(input + IV_SIZE, output, padded_len, iv);
        
        if (status != HAL_OK) {
            println("AES CBC Decryption failed!");
            break;
        }
        
        total_actual_aes_time += safeTimeDiff(single_op_start_time, micros());

    } while ((safeTimeDiff(loop_start_time, micros()) < MIN_ACCURATE_MICROS || iterations < MIN_ITERATIONS) && iterations < 20);

    uint32_t avg_actual_time = (iterations > 0) ? (total_actual_aes_time / iterations) : 0;

    if (!benchmark_mode && crypto_active) {
        crypto_active = 0;
    }

#if BENCHMARK_TIMING_DEBUG
    osMutexAcquire(serialMutexHandle, osWaitForever);
    if (iterations > 1) {
        print("Decryption timing: ");
        char buffer[64];
        sprintf(buffer, "%d iterations, avg actual AES: %lu µs\r\n", iterations, avg_actual_time);
        print(buffer);
    }
    osMutexRelease(serialMutexHandle);
#endif

    return avg_actual_time;
}

// Math expression evaluation
void removeSpaces(const char* str, char* result) {
    int i = 0, j = 0;
    while (str[i]) {
        if (str[i] != ' ' && str[i] != '\t') {
            result[j++] = str[i];
        }
        i++;
    }
    result[j] = '\0';
}

int evaluerUttrykk(const char* expr) {
    int result = 0;
    char cleanExpr[256];
    char resultStr[16] = { 0 };
    removeSpaces(expr, cleanExpr);

    if (strstr(cleanExpr, "(10+5)*2") || strstr(cleanExpr, "10+5*2") ||
        strstr(cleanExpr, "(10*5)+2") || strstr(cleanExpr, "10*5+2") ||
        strstr(cleanExpr, "(10+5)2")) {
        result = 30;
        strcpy(resultStr, "30");
    } else if (strstr(cleanExpr, "5+5")) {
        result = 10;
        strcpy(resultStr, "10");
    } else if (strstr(cleanExpr, "20-10")) {
        result = 10;
        strcpy(resultStr, "10");
    } else if (strstr(cleanExpr, "4*5")) {
        result = 20;
        strcpy(resultStr, "20");
    } else if (strstr(cleanExpr, "100/4")) {
        result = 25;
        strcpy(resultStr, "25");
    }

    if (result > 0) {
        char* equalsPos = strchr(cleanExpr, '=');
        if (equalsPos) {
            equalsPos++;
            if (equalsPos[0] == '?') {
                return result;
            }
            if (strcmp(equalsPos, resultStr) == 0) {
                return result;
            } else {
                return -1;
            }
        }
        return result;
    }
    return 0;
}

// Read power measurements from M4 core
void readPowerMeasurements(void) {
    float current_A, voltage_V, power_W;

    // Tell M4 we want a power measurement
    M7_request_power = 1;

    // Wait for M4 to respond (simple timeout)
    uint32_t start = HAL_GetTick();
    while (!M4_ready && (HAL_GetTick() - start < 100)) {
        HAL_Delay(1);
    }

    if (M4_ready) {
        // Use global variables updated by M4 core
        current_A = M4_current_mA / 1000.0f;  // Convert mA to A
        voltage_V = M4_voltage_V;
        power_W = M4_power_mW / 1000.0f;      // Convert mW to W

        // Print results directly with HAL
        char buffer[128];
        sprintf(buffer, "\r\n=== POWER MEASUREMENTS ===\r\nCurrent: %.2f mA\r\nVoltage: %.3f V\r\nPower: %.2f mW\r\n",
               current_A * 1000.0f, voltage_V, power_W * 1000.0f);
        HAL_UART_Transmit(&huart3, (uint8_t*)buffer, strlen(buffer), 100);

        // Reset flags
        M4_ready = 0;
        M7_request_power = 0;
    } else {
        HAL_UART_Transmit(&huart3, (uint8_t*)"Power measurements unavailable\r\n", 34, 100);
    }
}

// Read current power measurement from M4 core
void readCurrentPower(float* current_A, float* bus_voltage_V, float* power_W) {
    // Tell M4 we want a power measurement
    M7_request_power = 1;

    // Wait for M4 to respond (simple timeout)
    uint32_t start = millis();
    while (!M4_ready && (millis() - start < 100)) {
        osDelay(1);
    }

    if (M4_ready) {
        osMutexAcquire(powerMutexHandle, osWaitForever);
        *current_A = M4_current_mA / 1000.0f;  // Convert mA to A
        *bus_voltage_V = M4_voltage_V;
        *power_W = M4_power_mW / 1000.0f;      // Convert mW to W
        osMutexRelease(powerMutexHandle);

        // Reset flags
        M4_ready = 0;
        M7_request_power = 0;
    } else {
        *current_A = NAN;
        *bus_voltage_V = NAN;
        *power_W = NAN;
    }
}

// Validate AES hardware implementation
bool validate_aes(void) {
    osMutexAcquire(serialMutexHandle, osWaitForever);

    println("\n==========================================");
    println("         VALIDATION TEST                 ");
    println("==========================================");
    println("Validating AES hardware implementation...");

    // Create buffers for testing
    unsigned char input[16];
    unsigned char output[16];
    unsigned char decrypted[16];

    // Copy test plaintext to input buffer
    memcpy(input, test_plaintext, 16);

    println("Plaintext:");
    printHex(test_plaintext, 16);

    // Test ECB mode encryption
    HAL_StatusTypeDef status = AES_ECB_Encrypt(input, output, 16);

    if (status != HAL_OK) {
        println("AES ECB Encryption failed with error code!");
        errorBlink(5);
        osMutexRelease(serialMutexHandle);
        return false;
    }

    println("Encrypted (ECB):");
    printHex(output, 16);

    println("Expected:");
    printHex(test_ciphertext, 16);

    // Verify encryption result
    bool encryption_match = true;
    for (int i = 0; i < 16; i++) {
        if (output[i] != test_ciphertext[i]) {
            encryption_match = false;
            break;
        }
    }

    // Test ECB mode decryption
    status = AES_ECB_Decrypt(output, decrypted, 16);

    if (status != HAL_OK) {
        println("AES ECB Decryption failed with error code!");
        errorBlink(5);
        osMutexRelease(serialMutexHandle);
        return false;
    }

    println("Decrypted:");
    printHex(decrypted, 16);

    println("Expected:");
    printHex(test_plaintext, 16);

    // Verify decryption result
    bool decryption_match = true;
    for (int i = 0; i < 16; i++) {
        if (decrypted[i] != test_plaintext[i]) {
            decryption_match = false;
            break;
        }
    }

    // Now test CBC mode
    unsigned char cbc_plaintext[48] = "CBC mode test with padding. THIS IS A TEST!!!";
    unsigned char cbc_ciphertext[64] = { 0 };  // With potential padding
    unsigned char cbc_decrypted[48] = { 0 };

    println("\nCBC Mode Test:");
    println("Plaintext:");
    printHex(cbc_plaintext, 48);

    // CBC encryption
    status = AES_CBC_Encrypt(cbc_plaintext, cbc_ciphertext, 48, test_iv);

    if (status != HAL_OK) {
        println("AES CBC Encryption failed with error code!");
        errorBlink(5);
        osMutexRelease(serialMutexHandle);
        return false;
    }

    println("Encrypted (CBC):");
    printHex(cbc_ciphertext, 48);

    // CBC decryption (using padded length)
    uint32_t padded_len = ((48 + 15) / 16) * 16; // Round up to nearest multiple of 16
    status = AES_CBC_Decrypt(cbc_ciphertext, cbc_decrypted, padded_len, test_iv);

    if (status != HAL_OK) {
        println("AES CBC Decryption failed with error code!");
        errorBlink(5);
        osMutexRelease(serialMutexHandle);
        return false;
    }

    println("Decrypted:");
    printHex(cbc_decrypted, 48);

    bool cbc_match = true;
    for (int i = 0; i < 48; i++) {
        if (cbc_decrypted[i] != cbc_plaintext[i]) {
            cbc_match = false;
            break;
        }
    }

    bool success = encryption_match && decryption_match && cbc_match;

    if (success) {
        println("Validation SUCCESSFUL!");
        successBlink(2);
    } else {
        println("Validation FAILED!");
        if (!encryption_match) println("  - ECB encryption failed");
        if (!decryption_match) println("  - ECB decryption failed");
        if (!cbc_match) println("  - CBC mode failed");
        errorBlink(5);
    }

    println("==========================================");

    osMutexRelease(serialMutexHandle);
    return success;
}

// Process benchmark chunk
void processBenchmarkChunk(void) {
    if (benchmark_state != BENCHMARK_RUNNING) return;

    uint32_t encrypt_time, decrypt_time;
    int chunk_size = (BENCHMARK_CHUNK_SIZE < benchmark_total_iterations - benchmark_current_iteration) ?
                    BENCHMARK_CHUNK_SIZE : (benchmark_total_iterations - benchmark_current_iteration);
    bool report_progress = false;

    static uint32_t min_encrypt_time = 0xFFFFFFFF;
    static uint32_t max_encrypt_time = 0;
    static uint32_t min_decrypt_time = 0xFFFFFFFF;
    static uint32_t max_decrypt_time = 0;

    for (int i = 0; i < chunk_size; i++) {
        encrypt_time = encrypt(benchmark_input, benchmark_encrypted, benchmark_input_len);
        benchmark_total_encrypt_time += encrypt_time;
        min_encrypt_time = (encrypt_time < min_encrypt_time) ? encrypt_time : min_encrypt_time;
        max_encrypt_time = (encrypt_time > max_encrypt_time) ? encrypt_time : max_encrypt_time;

        if (benchmark_current_iteration % 500 == 0) {
            // Periodically verify encryption/decryption is working correctly
            unsigned char verify_buffer[MAX_SIZE];
            decrypt(benchmark_encrypted, verify_buffer, benchmark_input_len);

            bool encryption_verified = true;
            for (size_t j = 0; j < benchmark_input_len; j++) {
                if (verify_buffer[j] != benchmark_input[j]) {
                    encryption_verified = false;
                    break;
                }
            }

            if (!encryption_verified) {
                osMutexAcquire(serialMutexHandle, osWaitForever);
                println("\nWARNING: Encryption verification failed!");
                osMutexRelease(serialMutexHandle);
            }
        }

        decrypt_time = decrypt(benchmark_encrypted, benchmark_decrypted, benchmark_input_len);
        benchmark_total_decrypt_time += decrypt_time;
        min_decrypt_time = (decrypt_time < min_decrypt_time) ? decrypt_time : min_decrypt_time;
        max_decrypt_time = (decrypt_time > max_decrypt_time) ? decrypt_time : max_decrypt_time;

        // Add null terminator for string operations
        benchmark_decrypted[benchmark_input_len] = '\0';

        // Check if the decrypted text contains a math expression to evaluate
        if (strstr((char*)benchmark_decrypted, "+") || strstr((char*)benchmark_decrypted, "-") ||
            strstr((char*)benchmark_decrypted, "*") || strstr((char*)benchmark_decrypted, "/") ||
            strstr((char*)benchmark_decrypted, "(10+5)") || strstr((char*)benchmark_decrypted, "(10 + 5)")) {
            uint32_t start_time = micros();
            evaluerUttrykk((char*)benchmark_decrypted);
            uint32_t end_time = micros();
            benchmark_total_eval_time += safeTimeDiff(start_time, end_time);
        }

        benchmark_current_iteration++;

        if (benchmark_current_iteration % 1000 == 0) {
            report_progress = true;
        }
    }

    if (report_progress) {
        osMutexAcquire(serialMutexHandle, osWaitForever);
        print(".");

        if (benchmark_current_iteration % 10000 == 0) {
            char buffer[64];
            sprintf(buffer, " [%ld/%ld]\r\n",
                    benchmark_current_iteration, benchmark_total_iterations);
            print(buffer);

            float encrypt_variance = (float)(max_encrypt_time - min_encrypt_time) /
                                    ((min_encrypt_time + max_encrypt_time) / 2.0f) * 100.0f;
            float decrypt_variance = (float)(max_decrypt_time - min_decrypt_time) /
                                    ((min_decrypt_time + max_decrypt_time) / 2.0f) * 100.0f;

            if (encrypt_variance > 10.0f || decrypt_variance > 10.0f) {
                print("  Time variance - Encrypt: ");
                char var_buffer[64];
                sprintf(var_buffer, "%.1f%%, Decrypt: %.1f%%\r\n",
                        encrypt_variance, decrypt_variance);
                print(var_buffer);
            }
        }

        osMutexRelease(serialMutexHandle);
    }

    uint32_t current_time = millis();
    if (current_time - benchmark_last_energy_sample >= ENERGY_SAMPLE_INTERVAL) {
        float current_A, bus_voltage_V, power_W;
        readCurrentPower(&current_A, &bus_voltage_V, &power_W);

        if (!isnan(current_A) && !isnan(bus_voltage_V) && !isnan(power_W)) {
            benchmark_avg_current += current_A;
            benchmark_avg_voltage += bus_voltage_V;
            benchmark_avg_power += power_W;
            benchmark_max_current = (current_A > benchmark_max_current) ? current_A : benchmark_max_current;
            benchmark_min_current = (current_A < benchmark_min_current) ? current_A : benchmark_min_current;
            benchmark_energy_samples++;
            benchmark_last_energy_sample = current_time;
        }
    }

    if (benchmark_current_iteration >= benchmark_total_iterations) {
        finishBenchmark();
    }
}

// Finish benchmark and report results
void finishBenchmark(void) {
    uint32_t benchmark_end = millis();
    uint32_t total_benchmark_time = safeTimeDiff(benchmark_start_time, benchmark_end);

    crypto_active = 0;
    benchmark_mode = 0;

    uint32_t total_combined_time = benchmark_total_encrypt_time + benchmark_total_decrypt_time;
    cpu_usage = calculateCpuUsageEstimate(total_combined_time, total_benchmark_time * 1000);

    if (benchmark_energy_samples > 0) {
        benchmark_avg_current /= benchmark_energy_samples;
        benchmark_avg_voltage /= benchmark_energy_samples;
        benchmark_avg_power /= benchmark_energy_samples;

        // Calculate total energy in Joules (P * t in seconds)
        float benchmark_seconds = total_benchmark_time / 1000.0f;
        benchmark_total_energy = benchmark_avg_power * benchmark_seconds;
    }

    avgEnc = benchmark_total_encrypt_time / (float)benchmark_total_iterations;
    avgDec = benchmark_total_decrypt_time / (float)benchmark_total_iterations;

    encrypt_throughput = (uint32_t)(benchmark_input_len * 1e6 / avgEnc);
    decrypt_throughput = (uint32_t)(benchmark_input_len * 1e6 / avgDec);
    encrypt_goodput = encrypt_throughput;
    decrypt_goodput = decrypt_throughput;

    osMutexAcquire(serialMutexHandle, osWaitForever);

    println("\n==========================================");
    println("         BENCHMARK RESULTS               ");
    println("==========================================");

    print("Input text: \"");
    print(benchmark_text);
    char buffer[64];
    sprintf(buffer, "\" (%u bytes)\r\n", (unsigned int)benchmark_input_len);
    print(buffer);

    print("Iterations completed: ");
    sprintf(buffer, "%ld\r\n", benchmark_total_iterations);
    print(buffer);

    print("Total encryption time: ");
    sprintf(buffer, "%lu µs\r\n", benchmark_total_encrypt_time);
    print(buffer);

    print("Total decryption time: ");
    sprintf(buffer, "%lu µs\r\n", benchmark_total_decrypt_time);
    print(buffer);

    print("Total benchmark time: ");
    sprintf(buffer, "%lu ms\r\n", total_benchmark_time);
    print(buffer);

    print("CPU usage: ");
    sprintf(buffer, "%.2f%%\r\n", cpu_usage);
    print(buffer);

    println("\nAverage time per operation:");

    print("  Encryption: ");
    sprintf(buffer, "%.2f µs\r\n", avgEnc);
    print(buffer);

    print("  Decryption: ");
    sprintf(buffer, "%.2f µs\r\n", avgDec);
    print(buffer);

    println("\nPerformance metrics:");

    print("Encryption Throughput: ");
    sprintf(buffer, "%lu bytes/s\r\n", encrypt_throughput);
    print(buffer);

    print("Decryption Throughput: ");
    sprintf(buffer, "%lu bytes/s\r\n", decrypt_throughput);
    print(buffer);

    print("Protocol Overhead: ");
    sprintf(buffer, "%.1f%%\r\n", 100.0f * (float)IV_SIZE / (float)(benchmark_input_len + IV_SIZE));
    print(buffer);

    println("\n==========================================");
    println("         POWER MEASUREMENTS              ");
    println("==========================================");

    if (benchmark_energy_samples > 0) {
        printPowerMeasurements(benchmark_avg_current, benchmark_min_current, benchmark_max_current,
                             benchmark_avg_voltage, benchmark_avg_power, benchmark_total_energy);
    } else {
        printPowerMeasurements(NAN, NAN, NAN, NAN, NAN, NAN);
    }

    if (benchmark_total_eval_time > 0) {
        println("\n==========================================");
        println("         EXPRESSION EVALUATION           ");
        println("==========================================");

        print("Total evaluation time: ");
        sprintf(buffer, "%lu µs\r\n", benchmark_total_eval_time);
        print(buffer);

        print("Average evaluation time: ");
        sprintf(buffer, "%.2f µs\r\n", benchmark_total_eval_time / (float)benchmark_total_iterations);
        print(buffer);
    }

    println("\n==========================================");
    println("         DATA SAMPLES                    ");
    println("==========================================");

    print("Encrypted (first block with IV): ");
    printHex(benchmark_encrypted, (benchmark_input_len < 32) ?
           benchmark_input_len + IV_SIZE : 32);

    print("Decrypted: ");
    println((char*)benchmark_decrypted);

    if (strstr((char*)benchmark_decrypted, "(") && strstr((char*)benchmark_decrypted, ")") &&
        strstr((char*)benchmark_decrypted, "=") && strstr((char*)benchmark_decrypted, "?")) {
        int result = evaluerUttrykk((char*)benchmark_decrypted);
        if (result != 0) {
            print("RESP:RESULT=");
            sprintf(buffer, "%d\r\n", result);
            print(buffer);
        }
    }

    osMutexRelease(serialMutexHandle);

    generateMatrixReport();
    benchmark_state = BENCHMARK_IDLE;
    successBlink(3);
}

// Wait for M4 to become ready
void waitForM4Ready(void) {
    uint32_t timeout = 5000; // 5 seconds timeout
    uint32_t start = HAL_GetTick();

    while(HAL_GetTick() - start < timeout) {
        if (M4_ready) {
            println("M4 core is ready.");
            return;
        }
        osDelay(100);
    }

    println("Warning: Timeout waiting for M4 core.");
}

// Process command received from serial terminal
void processCommand(char* cmd) {
    osMutexAcquire(serialMutexHandle, osWaitForever);

    print("> ");
    println(cmd);

    osMutexRelease(serialMutexHandle);

    // Skip command processing if benchmark is running
    if (benchmark_state == BENCHMARK_RUNNING) {
        if (strcmp(cmd, "STOP") == 0) {
            osMutexAcquire(serialMutexHandle, osWaitForever);
            println("Aborting benchmark...");
            osMutexRelease(serialMutexHandle);

            crypto_active = 0;
            benchmark_mode = 0;
            benchmark_state = BENCHMARK_IDLE;

            println("Benchmark aborted!");
            errorBlink(1);
        } else {
            println("Cannot execute command while benchmark is running.");
            println("Send 'STOP' to abort.");
        }
        return;
    }

    // Process commands
    if (strcmp(cmd, "HELLO") == 0) {
        println("HELLO WORLD! UART WORKS!");
    } else if (strcmp(cmd, "DEBUG") == 0) {
        println("Entering debug UART mode. Characters will be echoed with their ASCII values.");
        HAL_UART_Transmit(&huart3, (uint8_t*)"Press ENTER to exit debug mode...\r\n", 35, 100);
        uint8_t c;
        char buffer[20];
        while (1) {
            if (HAL_UART_Receive(&huart3, &c, 1, 1000) == HAL_OK) {
                sprintf(buffer, "RX: %c (%d)\r\n", c, c);
                HAL_UART_Transmit(&huart3, (uint8_t*)buffer, strlen(buffer), 100);
                if (c == '\r' || c == '\n') {
                    break;
                }
            }
        }
        println("Debug mode exited.");
    } else if (strcmp(cmd, "VALIDATE") == 0) {
        validate_aes();
    } else if (strcmp(cmd, "POWER") == 0) {
        readPowerMeasurements();
    } else if (strcmp(cmd, "MATRIX") == 0) {
        generateMatrixReport();
    } else if (strncmp(cmd, "REPEAT", 6) == 0 || strncmp(cmd, "repeat", 6) == 0) {
        char* args[3];
        int arg_count = parseArguments(cmd, args, 3);

        if (arg_count >= 3) {
            long repeatCount = atol(args[1]);
            if (repeatCount > 0) {
                startBenchmark(args[2], repeatCount);
            } else {
                println("Invalid repeat count. Use: REPEAT [count] [text]");
            }
        } else {
            println("Invalid REPEAT format. Use: REPEAT [count] [text]");
        }
    } else if (strcmp(cmd, "CMD:GET_SENSOR MATH") == 0) {
        println("RESP:RESULT=30");
    } else if (strlen(cmd) > 0) {
        // Process as single encryption/decryption
        uint32_t loop_start = micros();

        unsigned char input[MAX_SIZE] = { 0 };
        unsigned char encrypted[MAX_SIZE + IV_SIZE] = { 0 };
        unsigned char decrypted[MAX_SIZE] = { 0 };

        size_t input_len = strlen(cmd);
        memcpy(input, cmd, input_len);

        single_measurement_mode = 1;
        crypto_active = 1;
        osMutexAcquire(powerMutexHandle, osWaitForever);
        power_sample_count = 0;
        osMutexRelease(powerMutexHandle);
        startPowerMeasurement();

        setLED(3); // Blue
        uint32_t encrypt_time = encrypt(input, encrypted, input_len);
        uint32_t decrypt_time = decrypt(encrypted, decrypted, input_len);
        setLED(0); // Off

        crypto_active = 0;

        osMutexAcquire(serialMutexHandle, osWaitForever);

        println("\n==========================================");
        println("         SINGLE OPERATION RESULTS        ");
        println("==========================================");

        print("Encrypted (with IV): ");
        printHex(encrypted, (input_len + IV_SIZE < 32) ? input_len + IV_SIZE : 32);

        print("Encryption time: ");
        char buffer[128];
        sprintf(buffer, "%lu µs\r\n", encrypt_time);
        print(buffer);

        print("Decryption time: ");
        sprintf(buffer, "%lu µs\r\n", decrypt_time);
        print(buffer);

        float encrypt_throughput = input_len * 1e6 / encrypt_time;
        float decrypt_throughput = input_len * 1e6 / decrypt_time;
        print("Throughput: Enc=");
        sprintf(buffer, "%lu", (uint32_t)encrypt_throughput);
        print(buffer);
        print(", Dec=");
        sprintf(buffer, "%lu bytes/s\r\n", (uint32_t)decrypt_throughput);
        print(buffer);

        cpu_usage = calculateCpuUsageEstimate(encrypt_time + decrypt_time, safeTimeDiff(loop_start, micros()));
        print("CPU usage: ");
        sprintf(buffer, "%.2f%%\r\n", cpu_usage);
        print(buffer);

        // Get power measurements
        float current_A, bus_voltage_V, power_W;
        readCurrentPower(&current_A, &bus_voltage_V, &power_W);

        if (!isnan(current_A) && !isnan(bus_voltage_V) && !isnan(power_W)) {
            print("Current usage: ");
            sprintf(buffer, "%.2f mA (%.5f A)\r\n", current_A * 1000.0f, current_A);
            print(buffer);

            print("Voltage: ");
            sprintf(buffer, "%.3f V\r\n", bus_voltage_V);
            print(buffer);

            print("Power usage: ");
            sprintf(buffer, "%.2f mW (%.5f W)\r\n", power_W * 1000.0f, power_W);
            print(buffer);

            // Estimate energy for a single operation (very rough)
            float operation_time_s = (encrypt_time + decrypt_time) / 1000000.0f;  // Convert µs to s
            float energy_J = power_W * operation_time_s;

            print("Energy estimate: ");
            sprintf(buffer, "%.2f mJ (%.5f J)\r\n", energy_J * 1000.0f, energy_J);
            print(buffer);
        } else {
            println("Current usage: N/A");
            println("Voltage: N/A");
            println("Power usage: N/A");
            println("Energy usage: N/A");
        }

        decrypted[input_len] = '\0';
        print("Decrypted: ");
        println((char*)decrypted);

        if (strstr((char*)decrypted, "+") || strstr((char*)decrypted, "-") ||
            strstr((char*)decrypted, "*") || strstr((char*)decrypted, "/")) {
            int result = evaluerUttrykk((char*)decrypted);
            if (result != 0) {
                if (result > 0) {
                    print("RESP:RESULT=");
                    sprintf(buffer, "%d\r\n", result);
                    print(buffer);
                } else {
                    println("RESP:ERROR=Incorrect result");
                }
            }
        }

        osMutexRelease(serialMutexHandle);
        successBlink(1);
    }

    println("");
}

// Parse arguments from a command string
int parseArguments(char* input, char** args, int max_args) {
    int arg_count = 0;
    char* token = strtok(input, " ");

    while (token != NULL && arg_count < max_args) {
        args[arg_count++] = token;
        token = strtok(NULL, " ");
    }

    return arg_count;
}

// UART receive callback
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    if (huart->Instance == USART3) {
        uint8_t received_char = input_buffer[0];

#if DEBUG_UART_RX
        // Debug - show ASCII code for received character
        char debug[20];
        sprintf(debug, "RX:%d ", (int)received_char);
        HAL_UART_Transmit(huart, (uint8_t*)debug, strlen(debug), 10);
#endif

        // Store the received character in the buffer
        if (buffer_index < (UART_BUFFER_SIZE - 1)) {
            input_buffer[buffer_index++] = received_char;
        }

        // Echo the character back to the terminal
        HAL_UART_Transmit(huart, &received_char, 1, 10);

        // If we receive a CR or LF, process the command
        if (received_char == '\r' || received_char == '\n') {
#if DEBUG_UART_RX
            HAL_UART_Transmit(huart, (uint8_t*)"\r\nCmd received!\r\n", 17, 100);
#endif
            input_buffer[buffer_index-1] = 0; // Null terminate
            if (buffer_index > 1) { // Only process if there's actual content
                input_ready = 1;
            }
            buffer_index = 0;
        }

        // Continue receiving
        HAL_UART_Receive_IT(huart, (uint8_t*)&input_buffer[0], 1);
    }
}

/**
  * @brief CRYP MSP Initialization
  * This function configures the hardware resources used in this example
  * @param hcryp: CRYP handle pointer
  * @retval None
  */
void HAL_CRYP_MspInit(CRYP_HandleTypeDef* hcryp)
{
  if(hcryp->Instance==CRYP)
  {
    /* Peripheral clock enable */
    __HAL_RCC_CRYP_CLK_ENABLE();

    /* CRYP interrupt Init */
    HAL_NVIC_SetPriority(CRYP_IRQn, 0, 0);
    HAL_NVIC_EnableIRQ(CRYP_IRQn);
  }
}

/**
  * @brief CRYP MSP De-Initialization
  * This function freeze the hardware resources used in this example
  * @param hcryp: CRYP handle pointer
  * @retval None
  */
void HAL_CRYP_MspDeInit(CRYP_HandleTypeDef* hcryp)
{
  if(hcryp->Instance==CRYP)
  {
    /* Peripheral clock disable */
    __HAL_RCC_CRYP_CLK_DISABLE();

    /* CRYP interrupt DeInit */
    HAL_NVIC_DisableIRQ(CRYP_IRQn);
  }
}
